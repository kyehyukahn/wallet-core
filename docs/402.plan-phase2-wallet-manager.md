**Goal:** Bitcoin xpub 기반 주소 체인 관리, UTXO 집계, 잔액/트랜잭션 히스토리 계산, persistence/recovery를 수행하는 `tw_wallet_manager` Rust crate를 구현한다.

**Design Spec:** `docs/300.design_bitcoin_improvemet.md` — Sections 10, 11, 12, 13

**Scope:** Phase 2만 다룬다. Phase 1 (SPV)은 완료, Phase 3 (Transaction improvement)과 Phase 4 (Multi-chain)는 별도 계획.

```rust
// rust/frameworks/tw_wallet_manager/src/error.rs
// SPDX-License-Identifier: Apache-2.0

use std::fmt;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WalletError {
    /// Address derivation failed.
    AddressDerivation(String),
    /// Transaction not found in wallet.
    TransactionNotFound,
    /// UTXO not found.
    UtxoNotFound,
    /// Insufficient balance for the requested operation.
    InsufficientBalance { available: u64, required: u64 },
    /// Persistence operation failed.
    Persistence(PersistenceError),
    /// Invalid wallet state (e.g. corrupted data).
    InvalidState(String),
}

impl fmt::Display for WalletError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            WalletError::AddressDerivation(msg) => write!(f, "address derivation: {}", msg),
            WalletError::TransactionNotFound => write!(f, "transaction not found"),
            WalletError::UtxoNotFound => write!(f, "utxo not found"),
            WalletError::InsufficientBalance { available, required } => {
                write!(f, "insufficient balance: {} available, {} required", available, required)
            }
            WalletError::Persistence(e) => write!(f, "persistence: {}", e),
            WalletError::InvalidState(msg) => write!(f, "invalid state: {}", msg),
        }
    }
}

impl std::error::Error for WalletError {}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PersistenceError {
    /// Storage backend reported an error.
    StorageError(String),
    /// Data could not be deserialized.
    DeserializationError(String),
    /// Atomic write failed (partial state may exist).
    AtomicWriteFailed(String),
}

impl fmt::Display for PersistenceError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            PersistenceError::StorageError(msg) => write!(f, "storage: {}", msg),
            PersistenceError::DeserializationError(msg) => write!(f, "deserialization: {}", msg),
            PersistenceError::AtomicWriteFailed(msg) => write!(f, "atomic write: {}", msg),
        }
    }
}

impl std::error::Error for PersistenceError {}

impl From<PersistenceError> for WalletError {
    fn from(e: PersistenceError) -> Self {
        WalletError::Persistence(e)
    }
}

pub type WalletResult<T> = Result<T, WalletError>;
```

```rust
// rust/frameworks/tw_wallet_manager/src/wallet_chain.rs
// SPDX-License-Identifier: Apache-2.0

/// Chain-specific wallet policy: address derivation rules, dust thresholds,
/// gap limits, and reorg depth hints.
///
/// Separated from SpvChain (network/sync rules) per design spec §6-4:
/// wallet state policies change independently from network protocol rules.
pub trait WalletChain {
    /// Number of consecutive unused external (receive) addresses to maintain.
    /// BIP44 standard gap limit is 20.
    fn external_gap_limit(&self) -> u32;

    /// Number of consecutive unused internal (change) addresses to maintain.
    fn internal_gap_limit(&self) -> u32;

    /// Minimum output value (satoshis) below which an output is considered dust
    /// at the given fee rate (satoshis per kB).
    /// Matches breadwallet-core BRWalletMinOutputAmount logic.
    fn dust_threshold(&self, fee_rate: u64) -> u64;

    /// Maximum expected reorg depth. Used to determine when to consider
    /// transactions "final" and to limit rescan depth.
    fn max_reorg_depth(&self) -> u32;

    /// BIP44 coin type for derivation path (e.g. 0 for Bitcoin, 1 for testnet).
    fn coin_type(&self) -> u32;
}
```

```rust
// rust/frameworks/tw_wallet_manager/src/bitcoin_wallet_chain.rs
// SPDX-License-Identifier: Apache-2.0

use crate::wallet_chain::WalletChain;

/// Bitcoin mainnet wallet chain policy.
pub struct BitcoinMainnetWallet;

impl WalletChain for BitcoinMainnetWallet {
    fn external_gap_limit(&self) -> u32 {
        20 // BIP44 standard
    }

    fn internal_gap_limit(&self) -> u32 {
        20
    }

    fn dust_threshold(&self, fee_rate: u64) -> u64 {
        // breadwallet-core formula: TX_MIN_OUTPUT_AMOUNT = feePerKb * 3 * (34 + 148) / 1000
        // where 34 = output size, 148 = input size
        // Minimum: 546 satoshis (Bitcoin Core default dust limit)
        let calculated = fee_rate * 3 * (34 + 148) / 1000;
        std::cmp::max(calculated, 546)
    }

    fn max_reorg_depth(&self) -> u32 {
        6 // 6 confirmations is the Bitcoin standard for finality
    }

    fn coin_type(&self) -> u32 {
        0 // BIP44 coin type for Bitcoin mainnet
    }
}
```

```rust
// SPDX-License-Identifier: Apache-2.0

use tw_hash::H256;
use tw_utxo::transaction::transaction_parts::OutPoint;

/// Transaction ID (alias for H256, double-SHA256 of raw tx).
pub type Txid = H256;

/// A single unspent transaction output tracked by the wallet.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct UtxoRecord {
    /// The outpoint (txid + vout index) identifying this UTXO.
    pub outpoint: OutPoint,
    /// Value in satoshis.
    pub value: u64,
    /// The scriptPubKey locking this output.
    pub script: Vec<u8>,
    /// Block height at which this UTXO was confirmed.
    /// `None` if unconfirmed (in mempool).
    pub block_height: Option<u32>,
    /// Whether this UTXO is from a change address (internal chain).
    pub is_change: bool,
}

/// A transaction tracked by the wallet.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WalletTransaction {
    /// Transaction ID.
    pub txid: Txid,
    /// Raw serialized transaction bytes.
    pub raw_tx: Vec<u8>,
    /// Block height where this transaction was confirmed.
    /// `None` if unconfirmed.
    pub block_height: Option<u32>,
    /// Block timestamp (seconds since epoch). 0 if unconfirmed.
    pub timestamp: u32,
    /// Amount received by wallet addresses (excluding change).
    pub amount_received: u64,
    /// Amount sent from wallet UTXOs (including change + fee).
    pub amount_sent: u64,
    /// Transaction fee (known only if all inputs are from this wallet).
    pub fee: Option<u64>,
}

impl WalletTransaction {
    /// Net amount from the wallet's perspective.
    /// Positive = received, negative = sent.
    pub fn net_amount(&self) -> i64 {
        self.amount_received as i64 - self.amount_sent as i64
    }

    /// Whether this transaction is confirmed.
    pub fn is_confirmed(&self) -> bool {
        self.block_height.is_some()
    }

    /// Number of confirmations given the current chain height.
    pub fn confirmations(&self, current_height: u32) -> u32 {
        match self.block_height {
            Some(h) if h <= current_height => current_height - h + 1,
            _ => 0,
        }
    }
}
```

- [ ] **Step 1: Create `address_manager.rs`**

```rust
// rust/frameworks/tw_wallet_manager/src/address_manager.rs
// SPDX-License-Identifier: Apache-2.0

use crate::error::{WalletError, WalletResult};
use crate::wallet_chain::WalletChain;
use bitcoin::bip32::{ChildNumber, ExtendedPubKey};
use bitcoin::secp256k1::Secp256k1;
use std::collections::HashSet;

/// Manages BIP32 address derivation chains (external + internal)
/// and enforces gap limit rules.
///
/// Mirrors breadwallet-core BRWallet's address chain management:
/// - External chain (m/0/i) for receive addresses
/// - Internal chain (m/1/i) for change addresses
/// - Gap limit: generate addresses until N consecutive unused addresses exist
pub struct AddressManager {
    xpub: ExtendedPubKey,
    secp: Secp256k1<bitcoin::secp256k1::All>,

    /// Next derivation index for external (receive) chain.
    external_cursor: u32,
    /// Next derivation index for internal (change) chain.
    internal_cursor: u32,

    /// All derived external addresses (as hex-encoded pubkey hashes or similar).
    external_addresses: Vec<String>,
    /// All derived internal addresses.
    internal_addresses: Vec<String>,

    /// Addresses known to have been used in transactions.
    used_addresses: HashSet<String>,
    /// All addresses ever generated (external + internal).
    all_addresses: HashSet<String>,

    /// Gap limit settings.
    external_gap_limit: u32,
    internal_gap_limit: u32,
}

impl AddressManager {
    /// Create a new AddressManager from an extended public key.
    ///
    /// The xpub should be at the account level (e.g. m/44'/0'/0')
    /// so that external = xpub/0/i and internal = xpub/1/i.
    pub fn new<C: WalletChain>(xpub: ExtendedPubKey, chain: &C) -> WalletResult<Self> {
        let secp = Secp256k1::new();
        let external_gap = chain.external_gap_limit();
        let internal_gap = chain.internal_gap_limit();

        let mut mgr = AddressManager {
            xpub,
            secp,
            external_cursor: 0,
            internal_cursor: 0,
            external_addresses: Vec::new(),
            internal_addresses: Vec::new(),
            used_addresses: HashSet::new(),
            all_addresses: HashSet::new(),
            external_gap_limit: external_gap,
            internal_gap_limit: internal_gap,
        };

        // Pre-generate addresses up to the gap limit
        mgr.ensure_gap_limit()?;

        Ok(mgr)
    }

    /// Derive a child public key at the given chain (0=external, 1=internal) and index.
    fn derive_address(&self, chain: u32, index: u32) -> WalletResult<String> {
        let chain_child = ChildNumber::from_normal_idx(chain)
            .map_err(|e| WalletError::AddressDerivation(format!("chain {}: {}", chain, e)))?;
        let index_child = ChildNumber::from_normal_idx(index)
            .map_err(|e| WalletError::AddressDerivation(format!("index {}: {}", index, e)))?;

        let chain_key = self
            .xpub
            .ckd_pub(&self.secp, chain_child)
            .map_err(|e| WalletError::AddressDerivation(format!("chain derivation: {}", e)))?;
        let child_key = chain_key
            .ckd_pub(&self.secp, index_child)
            .map_err(|e| WalletError::AddressDerivation(format!("index derivation: {}", e)))?;

        // Use the compressed public key hex as the address identifier.
        // In production, this would be encoded as a proper Bitcoin address
        // (P2PKH, P2WPKH, etc.) based on chain policy. For Phase 2,
        // we use the pubkey hash as the canonical identifier.
        let pubkey_bytes = child_key.public_key.serialize();
        let pubkey_hash = tw_hash::sha2::sha256(&pubkey_bytes);
        let hash160 = tw_hash::ripemd::ripemd_160(&pubkey_hash);
        Ok(tw_encoding::hex::encode(&hash160, false))
    }

    /// Ensure the gap limit is satisfied for both chains.
    /// Generates new addresses as needed.
    /// Returns the number of new addresses generated.
    pub fn ensure_gap_limit(&mut self) -> WalletResult<(u32, u32)> {
        let ext_new = self.ensure_chain_gap_limit(0)?;
        let int_new = self.ensure_chain_gap_limit(1)?;
        Ok((ext_new, int_new))
    }

    fn ensure_chain_gap_limit(&mut self, chain: u32) -> WalletResult<u32> {
        let gap_limit = if chain == 0 {
            self.external_gap_limit
        } else {
            self.internal_gap_limit
        };

        let cursor = if chain == 0 {
            &mut self.external_cursor
        } else {
            &mut self.internal_cursor
        };

        let addresses = if chain == 0 {
            &mut self.external_addresses
        } else {
            &mut self.internal_addresses
        };

        // Count consecutive unused addresses at the end
        let mut unused_tail = 0u32;
        for addr in addresses.iter().rev() {
            if self.used_addresses.contains(addr) {
                break;
            }
            unused_tail += 1;
        }

        let mut generated = 0u32;
        while unused_tail < gap_limit {
            let addr = self.derive_address(chain, *cursor)?;
            self.all_addresses.insert(addr.clone());
            addresses.push(addr);
            *cursor += 1;
            unused_tail += 1;
            generated += 1;
        }

        Ok(generated)
    }

    /// Mark an address as used (seen in a transaction).
    /// Returns true if the address was known, false if unknown.
    /// After marking, may need to call ensure_gap_limit() again.
    pub fn mark_address_used(&mut self, address: &str) -> bool {
        if self.all_addresses.contains(address) {
            self.used_addresses.insert(address.to_string());
            true
        } else {
            false
        }
    }

    /// Check if an address belongs to this wallet.
    pub fn contains_address(&self, address: &str) -> bool {
        self.all_addresses.contains(address)
    }

    /// Check if an address is from the internal (change) chain.
    pub fn is_change_address(&self, address: &str) -> bool {
        self.internal_addresses.iter().any(|a| a == address)
    }

    /// Get the first unused external (receive) address.
    pub fn receive_address(&self) -> Option<&str> {
        self.external_addresses
            .iter()
            .find(|a| !self.used_addresses.contains(a.as_str()))
            .map(|a| a.as_str())
    }

    /// Get the first unused internal (change) address.
    pub fn change_address(&self) -> Option<&str> {
        self.internal_addresses
            .iter()
            .find(|a| !self.used_addresses.contains(a.as_str()))
            .map(|a| a.as_str())
    }

    /// Total number of external addresses generated.
    pub fn external_count(&self) -> u32 {
        self.external_addresses.len() as u32
    }

    /// Total number of internal addresses generated.
    pub fn internal_count(&self) -> u32 {
        self.internal_addresses.len() as u32
    }

    /// All known addresses (for bloom filter population).
    pub fn all_addresses(&self) -> &HashSet<String> {
        &self.all_addresses
    }

    /// Cursors for persistence.
    pub fn cursors(&self) -> (u32, u32) {
        (self.external_cursor, self.internal_cursor)
    }

    /// Restore cursors from persistence and re-derive addresses.
    pub fn restore(
        xpub: ExtendedPubKey,
        external_cursor: u32,
        internal_cursor: u32,
        used_addresses: HashSet<String>,
        external_gap_limit: u32,
        internal_gap_limit: u32,
    ) -> WalletResult<Self> {
        let secp = Secp256k1::new();
        let mut mgr = AddressManager {
            xpub,
            secp,
            external_cursor: 0,
            internal_cursor: 0,
            external_addresses: Vec::new(),
            internal_addresses: Vec::new(),
            used_addresses,
            all_addresses: HashSet::new(),
            external_gap_limit,
            internal_gap_limit,
        };

        // Re-derive all addresses up to saved cursors
        for i in 0..external_cursor {
            let addr = mgr.derive_address(0, i)?;
            mgr.all_addresses.insert(addr.clone());
            mgr.external_addresses.push(addr);
        }
        mgr.external_cursor = external_cursor;

        for i in 0..internal_cursor {
            let addr = mgr.derive_address(1, i)?;
            mgr.all_addresses.insert(addr.clone());
            mgr.internal_addresses.push(addr);
        }
        mgr.internal_cursor = internal_cursor;

        // Ensure gap limit after restore
        mgr.ensure_gap_limit()?;

        Ok(mgr)
    }
}
```

```rust
use crate::types::UtxoRecord;
use std::collections::HashMap;
use tw_utxo::transaction::transaction_parts::OutPoint;

/// Tracks unspent transaction outputs for the wallet.
/// Uses HashMap<OutPoint, UtxoRecord> as primary store (design spec §10).
/// Sorting for coin selection is done at query time, not at storage time.
pub struct UtxoTracker {
    utxos: HashMap<OutPoint, UtxoRecord>,
}

impl Default for UtxoTracker {
    fn default() -> Self {
        Self::new()
    }
}

impl UtxoTracker {
    pub fn new() -> Self {
        UtxoTracker {
            utxos: HashMap::new(),
        }
    }

    /// Add or update a UTXO.
    pub fn add(&mut self, utxo: UtxoRecord) {
        self.utxos.insert(utxo.outpoint, utxo);
    }

    /// Remove a UTXO (when it's spent).
    pub fn remove(&mut self, outpoint: &OutPoint) -> Option<UtxoRecord> {
        self.utxos.remove(outpoint)
    }

    /// Check if a UTXO exists.
    pub fn contains(&self, outpoint: &OutPoint) -> bool {
        self.utxos.contains_key(outpoint)
    }

    /// Get a specific UTXO.
    pub fn get(&self, outpoint: &OutPoint) -> Option<&UtxoRecord> {
        self.utxos.get(outpoint)
    }

    /// Total wallet balance (sum of all UTXO values).
    pub fn balance(&self) -> u64 {
        self.utxos.values().map(|u| u.value).sum()
    }

    /// Confirmed balance only (UTXOs with a block_height).
    pub fn confirmed_balance(&self) -> u64 {
        self.utxos
            .values()
            .filter(|u| u.block_height.is_some())
            .map(|u| u.value)
            .sum()
    }

    /// Number of tracked UTXOs.
    pub fn count(&self) -> usize {
        self.utxos.len()
    }

    /// All UTXOs as a slice-like iterator.
    pub fn all(&self) -> impl Iterator<Item = &UtxoRecord> {
        self.utxos.values()
    }

    /// UTXOs sorted by value ascending (for coin selection).
    pub fn sorted_by_value_asc(&self) -> Vec<&UtxoRecord> {
        let mut sorted: Vec<_> = self.utxos.values().collect();
        sorted.sort_by_key(|u| u.value);
        sorted
    }

    /// UTXOs sorted by value descending (for coin selection).
    pub fn sorted_by_value_desc(&self) -> Vec<&UtxoRecord> {
        let mut sorted: Vec<_> = self.utxos.values().collect();
        sorted.sort_by(|a, b| b.value.cmp(&a.value));
        sorted
    }

    /// Clear all UTXOs (for rescan).
    pub fn clear(&mut self) {
        self.utxos.clear();
    }

    /// Bulk load from persistence.
    pub fn load(&mut self, utxos: Vec<UtxoRecord>) {
        for utxo in utxos {
            self.utxos.insert(utxo.outpoint, utxo);
        }
    }

    /// Export all UTXOs for persistence.
    pub fn export(&self) -> Vec<UtxoRecord> {
        self.utxos.values().cloned().collect()
    }

    /// Remove all UTXOs confirmed after a certain height (for reorg handling).
    /// Returns the removed UTXOs.
    pub fn remove_after_height(&mut self, height: u32) -> Vec<UtxoRecord> {
        let to_remove: Vec<OutPoint> = self
            .utxos
            .iter()
            .filter(|(_, u)| u.block_height.map(|h| h > height).unwrap_or(false))
            .map(|(k, _)| *k)
            .collect();

        to_remove
            .iter()
            .filter_map(|op| self.utxos.remove(op))
            .collect()
    }
}
```

```rust
    /// Register a new transaction. Returns true if it was new.
    pub fn register(&mut self, tx: WalletTransaction) -> bool {
        if self.transactions.contains_key(&tx.txid) {
            return false;
        }
        self.transactions.insert(tx.txid, tx);
        true
    }

    /// Remove a transaction. Returns the removed transaction.
    pub fn remove(&mut self, txid: &Txid) -> Option<WalletTransaction> {
        self.transactions.remove(txid)
    }

    /// Get a transaction by ID.

        self.transactions.get(txid)
    }

    /// Check if a transaction exists.
    pub fn contains(&self, txid: &Txid) -> bool {
        self.transactions.contains_key(txid)
    }

    /// Update block height and timestamp for transactions.
    /// Matches breadwallet-core BRWalletUpdateTransactions.
    pub fn update_confirmations(
        &mut self,
        txids: &[Txid],
        block_height: u32,
        timestamp: u32,
    ) -> Vec<Txid> {
        let mut updated = Vec::new();
        for txid in txids {
            if let Some(tx) = self.transactions.get_mut(txid) {
                tx.block_height = Some(block_height);
                tx.timestamp = timestamp;
                updated.push(*txid);
            }
        }
        updated
    }

    /// Mark all transactions confirmed after a given height as unconfirmed.
    /// Used for chain reorg handling.
    /// Matches breadwallet-core BRWalletSetTxUnconfirmedAfter.
    pub fn set_unconfirmed_after(&mut self, height: u32) -> Vec<Txid> {
        let mut affected = Vec::new();
        for tx in self.transactions.values_mut() {
            if let Some(h) = tx.block_height {
                if h > height {
                    tx.block_height = None;
                    tx.timestamp = 0;
                    affected.push(tx.txid);
                }
            }
        }
        affected
    }

    /// Get all transactions sorted by timestamp (newest first).
    pub fn all_sorted(&self) -> Vec<&WalletTransaction> {
        let mut txs: Vec<_> = self.transactions.values().collect();
        txs.sort_by(|a, b| b.timestamp.cmp(&a.timestamp));
        txs
    }

    /// Unconfirmed transactions only.
    pub fn unconfirmed(&self) -> Vec<&WalletTransaction> {
        self.transactions
            .values()
            .filter(|tx| tx.block_height.is_none())
            .collect()
    }

    /// Total amount received across all transactions.
    pub fn total_received(&self) -> u64 {
        self.transactions.values().map(|tx| tx.amount_received).sum()
    }

    /// Total amount sent across all transactions.
    pub fn total_sent(&self) -> u64 {
        self.transactions.values().map(|tx| tx.amount_sent).sum()
    }

    /// Number of tracked transactions.
    pub fn count(&self) -> usize {
        self.transactions.len()
    }

    /// Clear all transactions (for rescan).
    pub fn clear(&mut self) {
        self.transactions.clear();
    }

    /// Bulk load from persistence.
    pub fn load(&mut self, txs: Vec<WalletTransaction>) {
        for tx in txs {
            self.transactions.insert(tx.txid, tx);
        }
    }

    /// Export all transactions for persistence.
    pub fn export(&self) -> Vec<WalletTransaction> {
        self.transactions.values().cloned().collect()
    }
```

```rust
    /// Timestamp of the last synced block.

pub struct WalletState {
    /// External address derivation cursor.
    pub external_cursor: u32,
    /// Internal address derivation cursor.
    pub internal_cursor: u32,
    /// Set of addresses known to be used in transactions.
    pub used_addresses: HashSet<String>,

    pub utxos: Vec<UtxoRecord>,

    pub transactions: Vec<WalletTransaction>,
    /// Current wallet balance.
    pub balance: u64,

    pub fee_rate: u64,

    fn save_peer_cache(&self, peers: &[PeerInfo]) -> Result<(), PersistenceError>;

    fn save_chain_state(&self, state: &ChainState) -> Result<(), PersistenceError>;

    fn save_wallet_state(&self, state: &WalletState) -> Result<(), PersistenceError>;

    fn clear_wallet_state(&self) -> Result<(), PersistenceError>;

    /// Design spec §11-2b: prevents height/header inconsistency on crash.
    fn save_headers_batch(
        &self,
        headers: &[(u32, BlockHeader)],
        chain_state: &ChainState,
    ) -> Result<(), PersistenceError>;

    /// Atomically persist wallet state + chain state after filtered blocks.
    /// Design spec §11-2b: prevents balance inconsistency on crash.
    fn save_sync_progress(
        &self,
        chain_state: &ChainState,
        wallet_state: &WalletState,
    ) -> Result<(), PersistenceError>;
```

```rust
// rust/frameworks/tw_wallet_manager/src/recovery.rs
// SPDX-License-Identifier: Apache-2.0

use crate::error::WalletResult;
use crate::persistence::{ChainState, Persistence, WalletState};
use tw_spv::peer::PeerInfo;

/// Recovery actions determined by examining persisted state.
/// Design spec §11-3.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RecoveryAction {
    /// Everything is intact. Resume normal operation.
    ResumeNormal {
        chain_state: ChainState,
        wallet_state: WalletState,
        peers: Vec<PeerInfo>,
    },
    /// Wallet state is missing/corrupted but chain state is ok.
    /// Re-scan wallet from chain state, keeping headers.
    WalletRescan {
        chain_state: ChainState,
        peers: Vec<PeerInfo>,
    },
    /// Chain state is missing/corrupted.
    /// Full resync from last checkpoint.
    FullResync {
        peers: Vec<PeerInfo>,
    },
    /// No persisted state at all. Fresh start.
    FreshStart,
}

/// Examine persisted state and determine the appropriate recovery action.
pub fn determine_recovery_action(
    persistence: &dyn Persistence,
) -> WalletResult<RecoveryAction> {
    let peers = persistence
        .load_peer_cache()
        .unwrap_or_default();

    let chain_state = persistence.load_chain_state();
    let wallet_state = persistence.load_wallet_state();

    match (chain_state, wallet_state) {
        // Both intact
        (Ok(Some(cs)), Ok(Some(ws))) => {
            // Verify consistency: wallet height should not exceed chain height
            if ws.wallet_height > cs.last_height {
                // Wallet ahead of chain — inconsistent. Rescan wallet.
                Ok(RecoveryAction::WalletRescan {
                    chain_state: cs,
                    peers,
                })
            } else {
                Ok(RecoveryAction::ResumeNormal {
                    chain_state: cs,
                    wallet_state: ws,
                    peers,
                })
            }
        }
        // Chain state intact, wallet state missing
        (Ok(Some(cs)), _) => Ok(RecoveryAction::WalletRescan {
            chain_state: cs,
            peers,
        }),
        // Chain state missing or corrupt
        (Ok(None), _) | (Err(_), _) => {
            if peers.is_empty() {
                Ok(RecoveryAction::FreshStart)
            } else {
                Ok(RecoveryAction::FullResync { peers })
            }
        }
    }
}
```

```rust
use crate::error::WalletResult;
use crate::events::WalletEvent;

use std::collections::HashSet;
use tw_utxo::transaction::transaction_parts::OutPoint;

/// A stateful Bitcoin wallet tracking addresses, UTXOs, and transactions.
/// Design spec §10.
pub struct Wallet<C: WalletChain> {
    chain: C,
    addresses: AddressManager,
    utxos: UtxoTracker,
    txs: TxTracker,
    fee_rate: u64,
    wallet_height: u32,
}

impl<C: WalletChain> Wallet<C> {
    /// Create a new wallet from an extended public key.
    pub fn new(xpub: ExtendedPubKey, chain: C) -> WalletResult<Self> {
        let addresses = AddressManager::new(xpub, &chain)?;
        Ok(Wallet {
            chain,
            addresses,
            utxos: UtxoTracker::new(),
            txs: TxTracker::new(),
            fee_rate: 10000, // default 10 sat/byte
            wallet_height: 0,
        })
    }

    /// Current wallet balance (sum of all UTXOs).
    pub fn balance(&self) -> u64 {
        self.utxos.balance()
    }

    /// Confirmed balance only.
    pub fn confirmed_balance(&self) -> u64 {
        self.utxos.confirmed_balance()
    }

    /// Total received across all transactions.
    pub fn total_received(&self) -> u64 {
        self.txs.total_received()
    }

    /// Total sent across all transactions.
    pub fn total_sent(&self) -> u64 {
        self.txs.total_sent()
    }

    /// Current fee rate.
    pub fn fee_rate(&self) -> u64 {
        self.fee_rate
    }

    /// Set fee rate (e.g. from feefilter message).
    pub fn set_fee_rate(&mut self, fee_rate: u64) {
        self.fee_rate = fee_rate;
    }

    /// Dust threshold at current fee rate.
    pub fn dust_threshold(&self) -> u64 {
        self.chain.dust_threshold(self.fee_rate)
    }

    /// First unused receive address.
    pub fn receive_address(&self) -> Option<&str> {
        self.addresses.receive_address()
    }

    /// First unused change address.
    pub fn change_address(&self) -> Option<&str> {
        self.addresses.change_address()
    }

    /// Check if an address belongs to this wallet.
    pub fn contains_address(&self, address: &str) -> bool {
        self.addresses.contains_address(address)
    }

    /// All addresses for bloom filter.
    pub fn all_addresses(&self) -> &HashSet<String> {
        self.addresses.all_addresses()
    }

    /// Register a UTXO. Returns events if balance changed.
    pub fn add_utxo(&mut self, utxo: UtxoRecord) -> Vec<WalletEvent> {
        let old_balance = self.balance();
        self.utxos.add(utxo);
        let new_balance = self.balance();
        if new_balance != old_balance {
            vec![WalletEvent::BalanceChanged { new_balance }]
        } else {
            Vec::new()
        }
    }

    /// Spend a UTXO. Returns events if balance changed.
    pub fn spend_utxo(&mut self, outpoint: &OutPoint) -> Vec<WalletEvent> {
        let old_balance = self.balance();
        self.utxos.remove(outpoint);
        let new_balance = self.balance();
        if new_balance != old_balance {
            vec![WalletEvent::BalanceChanged { new_balance }]
        } else {
            Vec::new()
        }
    }

    /// Register a transaction. Returns events.
    pub fn register_transaction(&mut self, tx: WalletTransaction) -> Vec<WalletEvent> {
        let txid = tx.txid;
        if self.txs.register(tx) {
            vec![WalletEvent::TransactionAdded { txid }]
        } else {
            Vec::new()
        }
    }

    /// Mark an address as used and ensure gap limit.
    pub fn mark_address_used(&mut self, address: &str) -> WalletResult<Vec<WalletEvent>> {
        self.addresses.mark_address_used(address);
        let (ext_new, int_new) = self.addresses.ensure_gap_limit()?;
        if ext_new > 0 || int_new > 0 {
            Ok(vec![WalletEvent::AddressesGenerated {
                external_count: ext_new,
                internal_count: int_new,
            }])
        } else {
            Ok(Vec::new())
        }
    }

    /// Handle chain reorg: unconfirm txs and remove UTXOs after height.
    pub fn handle_reorg(&mut self, height: u32) -> Vec<WalletEvent> {
        let mut events = Vec::new();

        let affected_txs = self.txs.set_unconfirmed_after(height);
        for txid in &affected_txs {
            events.push(WalletEvent::TransactionUpdated {
                txid: *txid,
                block_height: None,
            });
        }

        let _removed_utxos = self.utxos.remove_after_height(height);
        let new_balance = self.balance();
        events.push(WalletEvent::BalanceChanged { new_balance });

        self.wallet_height = height;
        events
    }

    /// Export wallet state for persistence.
    pub fn export_state(&self) -> WalletState {
        let (ext_cursor, int_cursor) = self.addresses.cursors();
        WalletState {
            external_cursor: ext_cursor,
            internal_cursor: int_cursor,
            used_addresses: self.addresses.all_addresses().clone(),
            utxos: self.utxos.export(),
            transactions: self.txs.export(),
            balance: self.balance(),
            wallet_height: self.wallet_height,
            fee_rate: self.fee_rate,
        }
    }

    /// Get a transaction by ID.
    pub fn get_transaction(&self, txid: &Txid) -> Option<&WalletTransaction> {
        self.txs.get(txid)
    }

    /// All transactions sorted by timestamp.
    pub fn transactions(&self) -> Vec<&WalletTransaction> {
        self.txs.all_sorted()
    }

    /// UTXO count.
    pub fn utxo_count(&self) -> usize {
        self.utxos.count()
    }

    /// Transaction count.
    pub fn transaction_count(&self) -> usize {
        self.txs.count()
    }

    /// Current wallet height.
    pub fn wallet_height(&self) -> u32 {
        self.wallet_height
    }

    /// Set wallet height.
    pub fn set_wallet_height(&mut self, height: u32) {
        self.wallet_height = height;
    }

    /// Clear wallet data for rescan (keeps addresses).
    pub fn clear_for_rescan(&mut self) {
        self.utxos.clear();
        self.txs.clear();
        self.wallet_height = 0;
    }
}
```
