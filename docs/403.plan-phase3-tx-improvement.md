//
// Copyright © 2017 Trust Wallet.

use crate::transaction::transaction_parts::Amount;

pub mod dust_filter;

/// Estimated size of a typical transaction output (P2PKH).
const TX_OUTPUT_SIZE: Amount = 34;
/// Estimated size of a typical transaction input (compressed pubkey).
const TX_INPUT_SIZE: Amount = 148;

/// Transaction dust amount calculator.
#[derive(Clone, Copy)]
pub enum DustPolicy {
    /// Fixed dust threshold regardless of fee rate.
    FixedAmount(Amount),
    /// Dynamic dust threshold based on current fee rate.
    /// Matches breadwallet-core's BRWalletMinOutputAmount formula:
    /// threshold = max(fee_per_kb * 3 * (output_size + input_size) / 1000, min_amount)
    DynamicDust {
        /// Current fee rate in satoshis per kilobyte.
        fee_per_kb: Amount,
        /// Absolute minimum dust amount (Bitcoin Core default: 546 sat).
        min_amount: Amount,
    },
}

impl DustPolicy {
    pub fn dust_threshold(&self) -> Amount {
        match self {
            DustPolicy::FixedAmount(amount) => *amount,
            DustPolicy::DynamicDust {
                fee_per_kb,
                min_amount,
            } => {
                let calculated =
                    fee_per_kb * 3 * (TX_OUTPUT_SIZE + TX_INPUT_SIZE) / 1000;
                std::cmp::max(calculated, *min_amount)
            }
        }
    }
}
```

pub mod fee_estimator;

/// Standard fee policy.
pub enum FeePolicy {
    /// The amount of satoshis per vbyte ("satVb").
    FeePerVb(Amount),
    /// The amount of satoshis per kilobyte ("satKb").
    /// Matches breadwallet-core TX_FEE_PER_KB convention.
    FeePerKb(Amount),
}

impl FeePolicy {
    /// Convert to satoshis per vbyte for uniform calculation.
    /// FeePerKb is divided by 1000 (rounding up) to get per-byte rate.
    pub fn sat_per_vbyte(&self) -> Amount {
        match self {
            FeePolicy::FeePerVb(rate) => *rate,
            FeePolicy::FeePerKb(rate) => (*rate + 999) / 1000,
        }
    }
}

// SPDX-License-Identifier: Apache-2.0
//
// Copyright © 2017 Trust Wallet.

use crate::fee::FeePolicy;
use crate::transaction::transaction_interface::TransactionInterface;
use crate::transaction::transaction_parts::Amount;
use std::marker::PhantomData;
use tw_coin_entry::error::prelude::*;

pub trait FeeEstimator<Transaction> {
    fn estimate_fee(&self, tx: &Transaction) -> SigningResult<Amount>;
}

pub struct StandardFeeEstimator<Transaction> {
    policy: FeePolicy,
    _phantom: PhantomData<Transaction>,
}

impl<Transaction> StandardFeeEstimator<Transaction> {
    pub fn new(policy: FeePolicy) -> Self {
        StandardFeeEstimator {
            policy,
            _phantom: PhantomData,
        }
    }
}

impl<Transaction: TransactionInterface> FeeEstimator<Transaction>
    for StandardFeeEstimator<Transaction>
{
    fn estimate_fee(&self, tx: &Transaction) -> SigningResult<Amount> {
        let fee_per_vbyte = self.policy.sat_per_vbyte();

        let vsize = tx.vsize();
        Amount::try_from(vsize)
            .ok()
            .and_then(|vsize| vsize.checked_mul(fee_per_vbyte))
            .or_tw_err(SigningErrorType::Error_wrong_fee)
            .with_context(|| {
                format!("feePerVByte is too large: '{vsize} * {fee_per_vbyte}' overflow")
            })
    }
}

    /// Estimate the fee for sending a given amount.
    /// Matches breadwallet-core BRWalletFeeForTxAmount.
    ///
    /// Calculates by summing UTXOs needed to cover amount + estimated fee,
    /// accounting for a change output if applicable.
    pub fn estimate_fee_for_amount(&self, amount: u64) -> u64 {
        let fee_per_kb = self.fee_rate;
        let fee_per_byte = (fee_per_kb + 999) / 1000;

        // Get UTXOs sorted by value descending for greedy selection
        let sorted_utxos = self.utxos.sorted_by_value_desc();

        let mut input_total: u64 = 0;
        let mut input_count: usize = 0;

        for utxo in &sorted_utxos {
            input_total += utxo.value;
            input_count += 1;

            // Estimate tx size: 10 (overhead) + inputs * 148 + outputs * 34
            // 1 output for recipient, 1 for change
            let estimated_size = 10 + input_count * 148 + 2 * 34;
            let estimated_fee = (estimated_size as u64) * fee_per_byte as u64;

            if input_total >= amount + estimated_fee {
                return estimated_fee;
            }
        }

        // Not enough UTXOs — return fee for all inputs (max send scenario)
        let estimated_size = 10 + input_count * 148 + 1 * 34; // no change output
        (estimated_size as u64) * fee_per_byte as u64
    }

    /// Maximum amount that can be sent to a single address after fees.
    /// Matches breadwallet-core BRWalletMaxOutputAmount.
    pub fn max_spendable(&self) -> u64 {
        let total = self.balance();
        if total == 0 {
            return 0;
        }

        let fee_per_kb = self.fee_rate;
        let fee_per_byte = (fee_per_kb + 999) / 1000;

        let utxo_count = self.utxo_count();
        // All UTXOs as inputs, 1 output (recipient, no change)
        let estimated_size = 10 + utxo_count * 148 + 1 * 34;
        let estimated_fee = (estimated_size as u64) * fee_per_byte as u64;

        total.saturating_sub(estimated_fee)
    }

    /// Minimum output amount that is economically viable at current fee rate.
    /// Outputs below this are dust — spending them costs more than they're worth.
    /// Matches breadwallet-core BRWalletMinOutputAmount.
    pub fn min_output_amount(&self) -> u64 {
        let fee_per_kb = self.fee_rate;
        // breadwallet-core formula: fee_per_kb * 3 * (34 + 148) / 1000
        let calculated = fee_per_kb * 3 * (34 + 148) / 1000;
        std::cmp::max(calculated, 546) // Bitcoin Core minimum dust
    }
```

    /// Update fee rate from SPV peer's feefilter message (BIP133).
    /// The feefilter value is in satoshis per kilobyte.
    /// Only updates if the new rate is higher than current (peers report minimum).
    pub fn update_fee_rate_from_peer(&mut self, peer_fee_per_kb: u64) {
        let current = self.wallet.fee_rate();
        if peer_fee_per_kb > current {
            self.wallet.set_fee_rate(peer_fee_per_kb);
        }
    }

    /// Estimate fee for sending a given amount (delegates to wallet).
    pub fn estimate_fee(&self, amount: u64) -> u64 {
        self.wallet.estimate_fee_for_amount(amount)
    }

    /// Maximum sendable amount after fees (delegates to wallet).
    pub fn max_spendable(&self) -> u64 {
        self.wallet.max_spendable()
    }

    /// Minimum economically viable output amount (delegates to wallet).
    pub fn min_output_amount(&self) -> u64 {
        self.wallet.min_output_amount()
    }

// rust/tw_keypair/src/bip38/mod.rs
// SPDX-License-Identifier: Apache-2.0

pub mod scrypt_params;

use aes::cipher::{BlockDecrypt, BlockEncrypt, KeyInit};
use aes::Aes256;
use scrypt::scrypt;
use scrypt_params::*;
use tw_encoding::hex;
use tw_hash::sha2::sha256;

/// BIP38 error types.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Bip38Error {
    InvalidKey,
    InvalidPassphrase,
    ScryptFailed,
    InvalidFormat,
    ChecksumMismatch,
}

/// Check if a Base58Check-encoded string is a valid BIP38 encrypted key.
/// BIP38 keys start with 0x0142 (non-EC) or 0x0143 (EC-multiply).
pub fn is_valid_bip38(encoded: &[u8]) -> bool {
    if encoded.len() != 39 {
        return false;
    }
    encoded[0] == 0x01 && (encoded[1] == 0x42 || encoded[1] == 0x43)
}

/// Decrypt a BIP38 encrypted private key.
///
/// - `encrypted`: 39 bytes (prefix(2) + flags(1) + salt(4) + encrypted_key(32))
/// - `passphrase`: UTF-8 NFC normalized passphrase
///
/// Returns the 32-byte private key.
///
/// Algorithm matches breadwallet-core BRKeySetBIP38Key() for non-EC multiply mode.
pub fn decrypt(encrypted: &[u8], passphrase: &str) -> Result<[u8; 32], Bip38Error> {
    if encrypted.len() != 39 {
        return Err(Bip38Error::InvalidFormat);
    }
    if encrypted[0] != 0x01 || encrypted[1] != 0x42 {
        // Only non-EC multiply mode supported in Phase 3
        return Err(Bip38Error::InvalidFormat);
    }

    let flag = encrypted[2];
    let _compressed = (flag & 0x20) != 0;
    let salt = &encrypted[3..7];
    let encrypted_half1 = &encrypted[7..23];
    let encrypted_half2 = &encrypted[23..39];

    // Derive key using scrypt
    let params = scrypt::Params::new(14, SCRYPT_R, SCRYPT_P, SCRYPT_KEY_LEN)
        .map_err(|_| Bip38Error::ScryptFailed)?;
    let mut derived = [0u8; 64];
    scrypt(passphrase.as_bytes(), salt, &params, &mut derived)
        .map_err(|_| Bip38Error::ScryptFailed)?;

    let (derived_half1, derived_half2) = derived.split_at(32);

    // AES-256-ECB decrypt
    let cipher =
        Aes256::new(derived_half2.into());

    let mut block1 = [0u8; 16];
    block1.copy_from_slice(encrypted_half1);
    cipher.decrypt_block((&mut block1).into());

    let mut block2 = [0u8; 16];
    block2.copy_from_slice(encrypted_half2);
    cipher.decrypt_block((&mut block2).into());

    // XOR with derived_half1
    let mut privkey = [0u8; 32];
    for i in 0..16 {
        privkey[i] = block1[i] ^ derived_half1[i];
    }
    for i in 0..16 {
        privkey[16 + i] = block2[i] ^ derived_half1[16 + i];
    }

    Ok(privkey)
}

/// Encrypt a private key with BIP38 (non-EC multiply mode).
///
/// - `privkey`: 32-byte private key
/// - `passphrase`: UTF-8 NFC normalized passphrase
/// - `compressed`: whether to use compressed pubkey format
/// - `addr_hash`: first 4 bytes of SHA256d(Bitcoin address)
///
/// Returns 39 bytes: prefix(2) + flags(1) + addr_hash(4) + encrypted(32)
pub fn encrypt(
    privkey: &[u8; 32],
    passphrase: &str,
    compressed: bool,
    addr_hash: &[u8; 4],
) -> Result<[u8; 39], Bip38Error> {
    // Derive key using scrypt with addr_hash as salt
    let params = scrypt::Params::new(14, SCRYPT_R, SCRYPT_P, SCRYPT_KEY_LEN)
        .map_err(|_| Bip38Error::ScryptFailed)?;
    let mut derived = [0u8; 64];
    scrypt(passphrase.as_bytes(), addr_hash, &params, &mut derived)
        .map_err(|_| Bip38Error::ScryptFailed)?;

    let (derived_half1, derived_half2) = derived.split_at(32);

    // XOR privkey with derived_half1
    let mut xored = [0u8; 32];
    for i in 0..32 {
        xored[i] = privkey[i] ^ derived_half1[i];
    }

    // AES-256-ECB encrypt
    let cipher =
        Aes256::new(derived_half2.into());

    let mut block1 = [0u8; 16];
    block1.copy_from_slice(&xored[..16]);
    cipher.encrypt_block((&mut block1).into());

    let mut block2 = [0u8; 16];
    block2.copy_from_slice(&xored[16..]);
    cipher.encrypt_block((&mut block2).into());

    // Build result: 0x01 0x42 flags addr_hash encrypted
    let mut result = [0u8; 39];
    result[0] = 0x01;
    result[1] = 0x42;
    result[2] = if compressed { 0xe0 } else { 0xc0 };
    result[3..7].copy_from_slice(addr_hash);
    result[7..23].copy_from_slice(&block1);
    result[23..39].copy_from_slice(&block2);

    Ok(result)
}

mod tests {
    use super::*;

    #[test]
    fn test_is_valid_bip38() {
        let mut valid = [0u8; 39];
        valid[0] = 0x01;
        valid[1] = 0x42;
        assert!(is_valid_bip38(&valid));

        valid[1] = 0x43; // EC-multiply
        assert!(is_valid_bip38(&valid));

        valid[1] = 0x44; // invalid
        assert!(!is_valid_bip38(&valid));

        assert!(!is_valid_bip38(&[0u8; 10])); // wrong length
    }

    #[test]
    fn test_encrypt_decrypt_roundtrip() {
        let privkey = [0x42u8; 32];
        let passphrase = "TestPassphrase123";
        let addr_hash = [0xAA, 0xBB, 0xCC, 0xDD];

        let encrypted = encrypt(&privkey, passphrase, true, &addr_hash).unwrap();

        // Verify format
        assert_eq!(encrypted[0], 0x01);
        assert_eq!(encrypted[1], 0x42);
        assert_eq!(encrypted[2], 0xe0); // compressed flag
        assert_eq!(&encrypted[3..7], &addr_hash);

        // Decrypt and verify
        let decrypted = decrypt(&encrypted, passphrase).unwrap();
        assert_eq!(decrypted, privkey);
    }

    #[test]
    fn test_decrypt_wrong_passphrase() {
        let privkey = [0x42u8; 32];
        let addr_hash = [0xAA, 0xBB, 0xCC, 0xDD];

        let encrypted = encrypt(&privkey, "correct", true, &addr_hash).unwrap();
        let decrypted = decrypt(&encrypted, "wrong").unwrap();

        // Decryption "succeeds" but produces wrong key (BIP38 has no MAC)
        assert_ne!(decrypted, privkey);
    }

    #[test]
    fn test_invalid_format() {
        assert_eq!(
            decrypt(&[0u8; 10], "pass").unwrap_err(),
            Bip38Error::InvalidFormat
        );

        let mut bad = [0u8; 39];
        bad[0] = 0x01;
        bad[1] = 0x43; // EC-multiply not supported
        assert_eq!(
            decrypt(&bad, "pass").unwrap_err(),
            Bip38Error::InvalidFormat
        );
    }

    fn test_uncompressed_flag() {
        let privkey = [0x42u8; 32];
        let addr_hash = [0x11, 0x22, 0x33, 0x44];

        let encrypted = encrypt(&privkey, "test", false, &addr_hash).unwrap();
        assert_eq!(encrypted[2], 0xc0); // uncompressed flag

        let decrypted = decrypt(&encrypted, "test").unwrap();
        assert_eq!(decrypted, privkey);
    }
