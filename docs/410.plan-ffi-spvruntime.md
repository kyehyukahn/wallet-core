// rust/frameworks/tw_spv/src/tokio_runtime.rs
// SPDX-License-Identifier: Apache-2.0

//! Tokio-based implementation of SpvRuntime.
//! Gated behind the `tokio-runtime` feature flag.

#![cfg(feature = "tokio-runtime")]

use crate::runtime::{BoxFuture, SpvRuntime};
use std::time::SystemTime;

/// A concrete SpvRuntime backed by a tokio multi-thread runtime.
///
/// Owns its own tokio Runtime — apps do not need to provide one.
/// Worker threads are limited to 2 (sufficient for SPV peer connections).
pub struct TokioRuntime {
    rt: tokio::runtime::Runtime,
}

impl TokioRuntime {
    /// Create a new TokioRuntime with a multi-thread executor.
    pub fn new() -> std::io::Result<Self> {
        let rt = tokio::runtime::Builder::new_multi_thread()
            .worker_threads(2)
            .enable_all()
            .thread_name("spv-worker")
            .build()?;
        Ok(TokioRuntime { rt })
    }

    /// Run an async future to completion on this runtime.
    /// Used by FFI functions that need to call async code synchronously.
    pub fn block_on<F: std::future::Future>(&self, future: F) -> F::Output {
        self.rt.block_on(future)
    }

    /// Get a handle to the runtime for spawning tasks from outside.
    pub fn handle(&self) -> tokio::runtime::Handle {
        self.rt.handle().clone()
    }
}

impl SpvRuntime for TokioRuntime {
    fn spawn(&self, task: BoxFuture) {
        self.rt.spawn(task);
    }

    fn spawn_blocking(&self, task: BoxFuture) {
        self.rt.spawn(task);
    }

    fn now(&self) -> SystemTime {
        SystemTime::now()
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use std::sync::atomic::{AtomicBool, Ordering};
    use std::sync::Arc;

    #[test]
    fn test_tokio_runtime_creation() {
        let rt = TokioRuntime::new().unwrap();
        let now = rt.now();
        assert!(now.elapsed().unwrap().as_secs() < 1);
    }

    #[test]
    fn test_tokio_runtime_block_on() {
        let rt = TokioRuntime::new().unwrap();
        let result = rt.block_on(async { 42 });
        assert_eq!(result, 42);
    }

    #[test]
    fn test_tokio_runtime_spawn() {
        let rt = TokioRuntime::new().unwrap();
        let flag = Arc::new(AtomicBool::new(false));
        let flag_clone = flag.clone();

        rt.spawn(Box::pin(async move {
            flag_clone.store(true, Ordering::SeqCst);
        }));

        // Give the task time to complete
        rt.block_on(async {
            tokio::time::sleep(std::time::Duration::from_millis(50)).await;
        });

        assert!(flag.load(Ordering::SeqCst));
    }

    #[test]
    fn test_tokio_runtime_dns_resolve() {
        let rt = TokioRuntime::new().unwrap();
        let result = rt.block_on(async {
            tokio::net::lookup_host("localhost:8333").await
        });
        assert!(result.is_ok());
    }
}

/// Creates a new Bitcoin wallet manager from an xpub string.
///
/// \param xpub Non-null extended public key string (e.g. "xpub6...")
/// \return Nullable pointer to TWBitcoinWalletManager. Null if xpub is invalid.
#[no_mangle]
pub unsafe extern "C" fn tw_bitcoin_wallet_manager_create(
    xpub: *const TWString,
) -> *mut TWBitcoinWalletManagerImpl {
    let xpub_str = try_or_else!(TWString::from_ptr_as_ref(xpub), std::ptr::null_mut);
    let xpub_str = try_or_else!(xpub_str.as_str(), std::ptr::null_mut);

    let manager = try_or_else!(
        BitcoinWalletManager::new(extended_pub, persistence).ok(),
        std::ptr::null_mut
    );
    TWBitcoinWalletManagerImpl { manager }.into_ptr()
}

use tw_memory::ffi::RawPtrTrait;
use tw_misc::try_or_else;
use tw_spv::bitcoin_chain::BitcoinMainnet;
use tw_spv::chain::SpvChain;
use tw_spv::events::SyncCapability;
use tw_spv::peer_manager::PeerPool;
use tw_spv::sync_manager::SyncState;
use tw_spv::tokio_runtime::TokioRuntime;
use std::sync::atomic::{AtomicBool, Ordering};

/// Opaque handle wrapping SPV sync state for FFI.
pub struct TWBitcoinSpvSyncImpl {
    runtime: TokioRuntime,
    peer_pool: PeerPool,
    sync_state: SyncState,
    is_running: AtomicBool,
    chain: BitcoinMainnet,
}

impl RawPtrTrait for TWBitcoinSpvSyncImpl {}

/// Creates a new SPV sync instance for Bitcoin mainnet.
///
/// \param earliest_key_time Unix timestamp of wallet creation (for sync optimization)
/// \return Nullable pointer. Null if runtime creation fails.
#[no_mangle]
pub unsafe extern "C" fn tw_bitcoin_spv_sync_create(
    earliest_key_time: u32,
) -> *mut TWBitcoinSpvSyncImpl {
    let runtime = try_or_else!(TokioRuntime::new().ok(), std::ptr::null_mut);
    let chain = BitcoinMainnet;
    let sync_state = SyncState::new(&chain, earliest_key_time);

    TWBitcoinSpvSyncImpl {
        runtime,
        peer_pool: PeerPool::new(),
        sync_state,
        is_running: AtomicBool::new(false),
        chain,
    }
    .into_ptr()
}

/// Deletes SPV sync instance and shuts down the runtime.
#[no_mangle]
pub unsafe extern "C" fn tw_bitcoin_spv_sync_delete(
    sync: *mut TWBitcoinSpvSyncImpl,
) {
    let _ = TWBitcoinSpvSyncImpl::from_ptr(sync);
}

/// Starts SPV synchronization.
/// Resolves DNS seeds, connects to peers, begins header/block sync.
#[no_mangle]
pub unsafe extern "C" fn tw_bitcoin_spv_sync_start(
    sync: *mut TWBitcoinSpvSyncImpl,
) {
    let spv = try_or_else!(TWBitcoinSpvSyncImpl::from_ptr_as_mut(sync), ());

    if spv.is_running.load(Ordering::SeqCst) {
        return;
    }
    spv.is_running.store(true, Ordering::SeqCst);

    // Start sync state machine
    let _events = spv.sync_state.start_sync();

    // Resolve DNS seeds and populate peer pool
    let dns_seeds: Vec<String> = spv.chain.dns_seeds().iter().map(|s| s.to_string()).collect();
    let port = spv.chain.default_port();

    spv.runtime.spawn(Box::pin(async move {
        for seed in &dns_seeds {
            let addr = format!("{}:{}", seed, port);
            if let Ok(resolved) = tokio::net::lookup_host(&addr).await {
                for socket_addr in resolved {
                    // In a full implementation, PeerManager would connect here.
                    // For now, DNS resolution demonstrates the runtime works.
                    let _ = socket_addr;
                }
            }
        }
    }));
}

/// Stops SPV synchronization.
#[no_mangle]
pub unsafe extern "C" fn tw_bitcoin_spv_sync_stop(
    sync: *mut TWBitcoinSpvSyncImpl,
) {
    let spv = try_or_else!(TWBitcoinSpvSyncImpl::from_ptr_as_mut(sync), ());
    spv.is_running.store(false, Ordering::SeqCst);
    spv.sync_state.stop_sync();
}

/// Returns sync progress as a value between 0.0 and 1.0.
#[no_mangle]
pub unsafe extern "C" fn tw_bitcoin_spv_sync_progress(
    sync: *const TWBitcoinSpvSyncImpl,
) -> f64 {
    let spv = try_or_else!(TWBitcoinSpvSyncImpl::from_ptr_as_ref(sync), || 0.0);
    let target = spv.peer_pool.estimated_height();
    spv.sync_state.sync_progress(target)
}

/// Returns the current sync capability as an integer:
